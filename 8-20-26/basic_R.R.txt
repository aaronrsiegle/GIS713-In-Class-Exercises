# construct a new character vector recording coin flip outcomes
set.seed(42)
num_flips <- 100
coin_flips <- rep(NA, num_flips)
for(i in 1:num_flips){
  heads_or_tails <- rnorm(1)
  if(heads_or_tails >= 0){
    coin_flips[i] <- "H"
  }else{
    coin_flips[i] <- "T"
  }
}
coin_flips
sum(coin_flips == "H")

# what's the probability of getting this outcome if the coin was fair?
pbinom(q=sum(coin_flips == "H") - 1, size=num_flips, prob=0.5, lower.tail=FALSE)

# returns length of missing side of right triangle
myf <- function(a=NA, b=NA, c=NA){
  # check that we have at least two numbers
  if(sum(is.na(c(a, b, c))) > 1){
    print("must supply two of a, b, or c")
    return(NA)
  }
  if(is.na(a)) return(sqrt(c^2 - b^2))
  if(is.na(b)) return(sqrt(c^2 - a^2))
  if(is.na(c)) return(sqrt(a^2 + b^2))
}
myf(a=3, b=4)
myf(a=3, c=5)
myf(b=4, c=5)
myf(a=3)

# palindrome checker
myf <- function(phrase){
  phrase <- gsub(pattern=" ", rep="", phrase) # strip white space
  reversed <- paste(rev(unlist(strsplit(phrase,""))), collapse="") # reverse characters
  if(reversed == phrase){
    print("palindrome!")
  }else{
    print("sadly, not a palindrome")
  }
}
myf("racecar")
myf("dr awkward")
myf("Dr awkward")
